project(hwod LANGUAGES CXX)

add_executable(hwod src/main.cpp)
add_executable(hj1_last_word src/hj1_last_word.cpp)
add_executable(hj2_char_count src/hj2_char_count.cpp)
add_executable(hj3_random_unique src/hj3_random_unique.cpp)
add_executable(hj4_string_split src/hj4_string_split.cpp)
add_executable(hj5_hex_to_decimal src/hj5_hex_to_decimal.cpp)
add_executable(hj6_prime_factors src/hj6_prime_factors.cpp)
add_executable(hj7_approximation src/hj7_approximation.cpp)
add_executable(hj8_merge_records src/hj8_merge_records.cpp)
add_executable(hj9_unique_digits src/hj9_unique_digits.cpp)
add_executable(hj10_unique_characters src/hj10_unique_characters.cpp)
add_executable(hj13_sentence_reverse src/hj13_sentence_reverse.cpp)
target_compile_features(hwod PRIVATE cxx_std_17)
target_compile_features(hj1_last_word PRIVATE cxx_std_17)
target_compile_features(hj2_char_count PRIVATE cxx_std_17)
target_compile_features(hj3_random_unique PRIVATE cxx_std_17)
target_compile_features(hj4_string_split PRIVATE cxx_std_17)
target_compile_features(hj5_hex_to_decimal PRIVATE cxx_std_17)
target_compile_features(hj6_prime_factors PRIVATE cxx_std_17)
target_compile_features(hj7_approximation PRIVATE cxx_std_17)
target_compile_features(hj8_merge_records PRIVATE cxx_std_17)
target_compile_features(hj9_unique_digits PRIVATE cxx_std_17)
target_compile_features(hj10_unique_characters PRIVATE cxx_std_17)
target_compile_features(hj13_sentence_reverse PRIVATE cxx_std_17)
set_target_properties(hwod hj1_last_word hj2_char_count hj3_random_unique hj4_string_split hj5_hex_to_decimal hj6_prime_factors hj7_approximation hj8_merge_records hj9_unique_digits hj10_unique_characters hj13_sentence_reverse PROPERTIES CXX_EXTENSIONS OFF)
